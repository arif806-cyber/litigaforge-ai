2. Create src/hooks/useLanguage.js:

import { useState, useEffect } from "react";
import translations from "../i18n/index";

const COUNTRY_DEFAULT_LANG = {
  IN: "en", US: "en", GB: "en",
  AE: "en", AU: "en", CA: "en",
  SG: "en", DE: "de"
};

export function useLanguage(countryCode) {
  const storageKey = `lang_${countryCode}`;
  
  const [lang, setLang] = useState(() => {
    return localStorage.getItem(storageKey) ||
      COUNTRY_DEFAULT_LANG[countryCode] || "en";
  });

  useEffect(() => {
    // Set HTML lang attribute for 
    // accessibility and AI crawlers
    document.documentElement.lang = lang;
    // Set RTL for Arabic
    document.documentElement.dir =
      lang === "ar" ? "rtl" : "ltr";
  }, [lang]);

  const switchLang = (newLang) => {
    localStorage.setItem(storageKey, newLang);
    setLang(newLang);
  };

  const countryTranslations = 
    translations[countryCode] || 
    translations["IN"];
    
  const t = countryTranslations[lang] ||
    countryTranslations["en"] || {};

  const availableLangs = Object.keys(
    countryTranslations
  );

  return { lang, switchLang, t, availableLangs };
}

3. Create src/components/LanguageSwitcher.jsx:

import { useLanguage } from "../hooks/useLanguage";

const LANG_LABELS = {
  en: "English",
  hi: "हिंदी",
  te: "తెలుగు",
  ar: "العربية",
  de: "Deutsch",
  fr: "Français",
  es: "Español"
};

const LANG_FLAGS = {
  en: "🇬🇧", hi: "🇮🇳", te: "🇮🇳",
  ar: "🇦🇪", de: "🇩🇪", fr: "🇫🇷",
  es: "🇪🇸"
};

export default function LanguageSwitcher({ 
  countryCode 
}) {
  const { lang, switchLang, availableLangs } =
    useLanguage(countryCode);

  if (availableLangs.length <= 1) return null;

  return (
    <div className="flex items-center gap-1">
      {availableLangs.map(l => (
        <button
          key={l}
          onClick={() => switchLang(l)}
          className={`flex items-center gap-1 
          px-3 py-1.5 rounded-full text-xs 
          font-medium transition-all
          ${l === lang
            ? "bg-yellow-500 text-black shadow-lg"
            : "bg-gray-700 text-gray-300 hover:bg-gray-600"
          }`}
        >
          <span>{LANG_FLAGS[l]}</span>
          <span>{LANG_LABELS[l]}</span>
        </button>
      ))}
    </div>
  );
}
