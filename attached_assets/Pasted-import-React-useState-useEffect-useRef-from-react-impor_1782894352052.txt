function Glass({ className = "", style = {}, children, ...rest }) {
  return (
    <div
      className={`rounded-2xl backdrop-blur-md ${className}`}
      style={{ backgroundColor: "rgba(255,255,255,0.035)", border: `1px solid ${surfaceBorder}`, ...style }}
      {...rest}
    >
      {children}
    </div>
  );
}

function Ring({ value, color, size = 74, label, sub }) {
  const [animated, setAnimated] = useState(0);
  const r = (size - 10) / 2;
  const c = 2 * Math.PI * r;
  useEffect(() => {
    const t = setTimeout(() => setAnimated(value), 150);
    return () => clearTimeout(t);
  }, [value]);
  return (
    <div className="flex items-center gap-3">
      <div className="relative" style={{ width: size, height: size }}>
        <svg width={size} height={size} className="-rotate-90">
          <circle cx={size / 2} cy={size / 2} r={r} stroke="rgba(255,255,255,0.08)" strokeWidth="7" fill="none" />
          <circle
            cx={size / 2} cy={size / 2} r={r} stroke={color} strokeWidth="7" fill="none"
            strokeLinecap="round"
            strokeDasharray={c}
            strokeDashoffset={c - (animated / 100) * c}
            style={{ transition: "stroke-dashoffset 1.1s cubic-bezier(.4,0,.2,1)", filter: `drop-shadow(0 0 6px ${color}99)` }}
          />
        </svg>
        <span className="absolute inset-0 flex items-center justify-center text-sm font-bold" style={{ color: textPrimary, fontFamily: mono }}>
          {value}%
        </span>
      </div>
      <div>
        <p className="text-xs font-semibold" style={{ color: textPrimary }}>{label}</p>
        <p className="text-[10px]" style={{ color: textSoft }}>{sub}</p>
      </div>
    </div>
  );
}

function Sparkline({ color }) {
  return (
    <div style={{ width: "100%", height: 34 }}>
      <ResponsiveContainer>
        <AreaChart data={sparkData}>
          <defs>
            <linearGradient id={`grad-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
              <stop offset="0%" stopColor={color} stopOpacity={0.5} />
              <stop offset="100%" stopColor={color} stopOpacity={0} />
            </linearGradient>
          </defs>
          <Area type="monotone" dataKey="v" stroke={color} strokeWidth={2} fill={`url(#grad-${color.replace("#", "")})`} />
        </AreaChart>
      </ResponsiveContainer>
    </div>
  );
}

function Typewriter({ text, accent }) {
  const [shown, setShown] = useState("");
  useEffect(() => {
    setShown("");
    let i = 0;
    const id = setInterval(() => {
      i++;
      setShown(text.slice(0, i));
      if (i >= text.length) clearInterval(id);
    }, 22);
    return () => clearInterval(id);
  }, [text]);
  return (
    <p className="text-xs leading-relaxed" style={{ color: textPrimary, fontFamily: sans, minHeight: "2.6em" }}>
      {shown}
      <span className="inline-block w-1 h-3 ml-0.5 align-middle animate-pulse" style={{ backgroundColor: accent }} />
    </p>
  );
}

                {advocatePractice.map((t) => <FileTab key={t.title} {...t} accent={{ solid: emerald, dim: emeraldDim }} />)}
              </div>
            </section>

            <section>
              <h2 className="text-[11px] font-semibold tracking-widest uppercase mb-3" style={{ color: textSoft }}>Activity</h2>
              <Glass className="p-4">
                <TimelineRow icon={Zap} title="New proposal received" time="1 hour ago" color={violet} />
                <TimelineRow icon={Wallet} title="Payout of ₹8,500 processed" time="Yesterday" color={emerald} />
                <TimelineRow icon={Star} title="Client left a 5★ review" time="2 days ago" color={amber} />
              </Glass>
            </section>
          </main>
        )}

        {/* bottom nav */}
        <nav
          className="fixed bottom-0 z-20 flex items-end justify-around px-2 pt-2"
          style={{
            width: "100%", maxWidth: 420, backgroundColor: "rgba(10,11,16,0.9)", backdropFilter: "blur(12px)",
            borderTop: `1px solid ${surfaceBorder}`, paddingBottom: "max(0.5rem, env(safe-area-inset-bottom))",
          }}
        >
          {role === "client" ? (
            <>
              <NavItem icon={Briefcase} label="Dashboard" active accent={accent.solid} />
              <NavItem icon={Inbox} label="Cases" accent={accent.solid} />
              <div className="flex-1 flex justify-center">
                <button className="-mt-6 w-12 h-12 rounded-full flex items-center justify-center active:scale-95 transition-transform" style={{ backgroundColor: amber, boxShadow: `0 4px 16px ${amber}77` }}>
                  <Plus size={22} color="#0A0B10" strokeWidth={2.8} />
                </button>
              </div>
              <NavItem icon={MessageSquareText} label="Messages" accent={accent.solid} />
              <NavItem icon={MoreHorizontal} label="More" accent={accent.solid} onClick={() => setMoreOpen(true)} />
            </>
          ) : (
            <>
              <NavItem icon={Briefcase} label="Dashboard" active accent={accent.solid} />
              <NavItem icon={ClipboardList} label="Proposals" accent={accent.solid} />
              <div className="flex-1 flex justify-center">
                <button className="-mt-6 w-12 h-12 rounded-full flex items-center justify-center active:scale-95 transition-transform" style={{ backgroundColor: emerald, boxShadow: `0 4px 16px ${emerald}77` }}>
                  <Wallet size={20} color="#0A0B10" strokeWidth={2.6} />
                </button>
              </div>
              <NavItem icon={MessageSquareText} label="Messages" accent={accent.solid} />
              <NavItem icon={MoreHorizontal} label="More" accent={accent.solid} onClick={() => setMoreOpen(true)} />
            </>
          )}
        </nav>

        <MoreSheet role={role} open={moreOpen} onClose={() => setMoreOpen(false)} />
      </div>
    </div>
  );
}
