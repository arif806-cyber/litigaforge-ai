STEP 3 — Create backend/llm/legal_llm.py
from litellm import completion
from backend.llm.config import (
    LLM_MODEL, LLM_MAX_TOKENS, LEGAL_SYSTEM_PROMPT
)

def ask_legal_question(
    question: str,
    context: str = "",
    language: str = "en"
) -> str:
    messages = [
        {"role": "system", "content": LEGAL_SYSTEM_PROMPT}
    ]
    if context:
        messages.append({
            "role": "system",
            "content": f"Relevant case context:\n{context}"
        })
    messages.append({"role": "user", "content": question})
    
    response = completion(
        model=LLM_MODEL,
        messages=messages,
        max_tokens=LLM_MAX_TOKENS
    )
    return response.choices[0].message.content

def summarize_judgment(judgment_text: str) -> dict:
    prompt = """
    Analyze this Indian court judgment and return a 
    JSON object with exactly these keys:
    {
      "summary_en": "3 sentence plain English summary",
      "summary_hi": "3 sentence plain Hindi summary",
      "outcome": "one of: Upheld/Dismissed/Landmark/Remanded",
      "acts_cited": ["list", "of", "acts", "and", "sections"],
      "area_of_law": "primary legal area",
      "key_principle": "the main legal principle in one sentence"
    }
    
    Return ONLY the JSON object. No markdown, no backticks,
    no explanation. Just the raw JSON.
    
    Judgment text:
    """ + judgment_text[:4000]
    
    messages = [
        {"role": "system", "content": LEGAL_SYSTEM_PROMPT},
        {"role": "user", "content": prompt}
    ]
    response = completion(
        model=LLM_MODEL,
        messages=messages,
        max_tokens=LLM_MAX_TOKENS
    )
    import json
    raw = response.choices[0].message.content
    clean = raw.replace("```json", "").replace("```", "").strip()
    return json.loads(clean)

def translate_to_hindi(text: str) -> str:
    messages = [
        {"role": "system", "content": 
         "Translate this legal text to simple Hindi. "
         "Keep legal terms in English but explain them "
         "in brackets in Hindi."},
        {"role": "user", "content": text}
    ]
    response = completion(
        model=LLM_MODEL,
        messages=messages,
        max_tokens=LLM_MAX_TOKENS
    )
    return response.choices[0].message.content
