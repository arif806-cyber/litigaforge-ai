PART D — REWRITE main.py
═══════════════════════════════════════

After moving all routes out, main.py should contain ONLY:

   import os
   from fastapi import FastAPI
   from fastapi.middleware.cors import CORSMiddleware
   from slowapi import Limiter, _rate_limit_exceeded_handler
   from slowapi.util import get_remote_address
   from slowapi.errors import RateLimitExceeded
   from database import get_pool, close_pool

   from routers import (
       auth, forge, subscription,
       matching, chat, community,
       watch, alerts, admin
   )

   app = FastAPI(
       title="LitigaForge AI",
       description="Legal AI Platform for Telangana & Andhra Pradesh",
       version="1.0.0",
       docs_url="/litigaforge/docs",
       redoc_url="/litigaforge/redoc"
   )

   limiter = Limiter(key_func=get_remote_address)
   app.state.limiter = limiter
   app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)

   allowed_origins = [
       os.environ.get("FRONTEND_URL", "http://localhost:5173"),
       "http://localhost:5173",
       "http://localhost:4173",
   ]
   app.add_middleware(
       CORSMiddleware,
       allow_origins=allowed_origins,
       allow_credentials=True,
       allow_methods=["GET", "POST", "PUT", "DELETE", "OPTIONS"],
       allow_headers=["Authorization", "Content-Type", "Accept"],
       max_age=600,
   )

   @app.on_event("startup")
   async def startup():
       assert os.environ.get("SESSION_SECRET"), \
           "SESSION_SECRET is required"
       assert os.environ.get("DATABASE_URL"), \
           "DATABASE_URL is required"
       await get_pool()
       print("✓ LitigaForge AI started")

   @app.on_event("shutdown")
   async def shutdown():
       await close_pool()

   app.include_router(auth.router)
   app.include_router(forge.router)
   app.include_router(subscription.router)
   app.include_router(matching.router)
   app.include_router(chat.router)
   app.include_router(community.router)
   app.include_router(watch.router)
   app.include_router(alerts.router)
   app.include_router(admin.router)

main.py must have zero route definitions after this refactor.
All routes live only in their router files.
