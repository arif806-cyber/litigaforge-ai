import os
import httpx
import asyncio
from typing import Optional

IK_TOKEN = os.getenv("INDIANKANOON_TOKEN")
IK_BASE = "https://api.indiankanoon.org"
HEADERS = {"Authorization": f"Token {IK_TOKEN}"}

# Cost tracker — log every API call
# Search = ₹0.50, Document = ₹0.20, Fragment = ₹0.05

async def search_judgments(
    query: str,
    pagenum: int = 0,
    court: str = None
) -> dict:
    """Search Indian Kanoon. Cost: ₹0.50 per call."""
    params = {"formInput": query, "pagenum": pagenum}
    if court:
        params["court"] = court
    async with httpx.AsyncClient(timeout=30) as client:
        r = await client.post(
            f"{IK_BASE}/search/",
            data=params,
            headers=HEADERS
        )
        r.raise_for_status()
        return r.json()

async def get_document(doc_id: str) -> dict:
    """Fetch full judgment. Cost: ₹0.20 per call."""
    await asyncio.sleep(2)  # Rate limit: 1 req/2 sec
    async with httpx.AsyncClient(timeout=30) as client:
        r = await client.get(
            f"{IK_BASE}/doc/{doc_id}/",
            headers=HEADERS
        )
        r.raise_for_status()
        return r.json()

async def get_doc_metadata(doc_id: str) -> dict:
    """Fetch metadata only. Cost: ₹0.02 per call."""
    async with httpx.AsyncClient(timeout=30) as client:
        r = await client.get(
            f"{IK_BASE}/docmeta/{doc_id}/",
            headers=HEADERS
        )
        r.raise_for_status()
        return r.json()
