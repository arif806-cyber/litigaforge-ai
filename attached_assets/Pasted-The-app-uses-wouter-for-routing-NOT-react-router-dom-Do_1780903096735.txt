The app uses wouter for routing, NOT react-router-dom.
Do NOT install or import react-router-dom anywhere.
Use wouter throughout for all routing.

=== PART 1: APP ROUTES (using wouter) ===

In src/App.jsx, update routing using wouter:

import { Switch, Route, useLocation } 
  from "wouter";
import CountryLanding from 
  "./pages/CountryLanding";
import AutoRedirect from 
  "./components/AutoRedirect";

const VALID_COUNTRIES = [
  "in","us","gb","ae","au","ca","sg","de"
];

// Inside your app JSX, replace current 
// router with:

<Switch>
  {/* Root → auto detect and redirect */}
  <Route path="/" component={AutoRedirect} />

  {/* Country pages */}
  <Route path="/:countryCode">
    {(params) => {
      const code = params.countryCode
        ?.toLowerCase();
      if (!VALID_COUNTRIES.includes(code)) {
        return <AutoRedirect />;
      }
      return (
        <CountryLanding 
          countryCode={code.toUpperCase()} 
        />
      );
    }}
  </Route>

  {/* Country sub-routes e.g. /de/legal */}
  <Route path="/:countryCode/:section">
    {(params) => {
      const code = params.countryCode
        ?.toLowerCase();
      if (!VALID_COUNTRIES.includes(code)) {
        return <AutoRedirect />;
      }
      return (
        <CountryLanding
          countryCode={code.toUpperCase()}
          section={params.section}
        />
      );
    }}
  </Route>
</Switch>

=== PART 2: AUTO REDIRECT COMPONENT ===

Create src/components/AutoRedirect.jsx:

import { useEffect, useState } from "react";
import { useLocation } from "wouter";

const VALID_COUNTRIES = [
  "in","us","gb","ae","au","ca","sg","de"
];

export default function AutoRedirect() {
  const [, setLocation] = useLocation();
  const [detecting, setDetecting] = 
    useState(true);

  useEffect(() => {
    // Check manual override first
    const override = localStorage.getItem(
      "country_override"
    );
    if (override && VALID_COUNTRIES
      .includes(override.toLowerCase())) {
      setLocation(`/${override.toLowerCase()}`);
      return;
    }

    // Auto detect via IP
    fetch("/api/country-detect")
      .then(r => r.json())
      .then(data => {
        const code = data.country_code
          ?.toLowerCase() || "in";
        const valid = VALID_COUNTRIES
          .includes(code) ? code : "in";
        setLocation(`/${valid}`);
      })
      .catch(() => setLocation("/in"))
      .finally(() => setDetecting(false));
  }, []);

  return (
    <div className="min-h-screen bg-gray-950 
    flex flex-col items-center 
    justify-center gap-4">
      <div className="text-4xl animate-pulse">
        🌍
      </div>
      <p className="text-white text-lg 
      font-medium">
        Detecting your location...
      </p>
      <p className="text-gray-500 text-sm">
        You'll be redirected to your 
        country's legal platform
      </p>
    </div>
  );
}
