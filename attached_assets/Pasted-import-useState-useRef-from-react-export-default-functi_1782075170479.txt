import { useState, useRef } from "react";

export default function IncomingCaseFeed({ cases, onAccept, onDecline }) {
  const [index, setIndex] = useState(0);
  const [dragX, setDragX] = useState(0);
  const startX = useRef(0);

  const current = cases[index];
  if (!current) return <EmptyState />;

  const handleStart = (e) => { startX.current = e.touches ? e.touches[0].clientX : e.clientX; };
  const handleMove = (e) => {
    const x = e.touches ? e.touches[0].clientX : e.clientX;
    setDragX(x - startX.current);
  };
  const handleEnd = () => {
    if (dragX > 100) { onAccept(current.id); advance(); }
    else if (dragX < -100) { onDecline(current.id); advance(); }
    setDragX(0);
  };
  const advance = () => setIndex((i) => i + 1);

  const rotation = dragX / 20;
  const bgTint = dragX > 50 ? "bg-green-50" : dragX < -50 ? "bg-red-50" : "bg-white";

  return (
    <div className="relative h-80 px-4">
      <div
        className={`absolute inset-x-4 top-0 rounded-2xl shadow-lg p-4 ${bgTint} transition-colors`}
        style={{ transform: `translateX(${dragX}px) rotate(${rotation}deg)`, touchAction: "none" }}
        onMouseDown={handleStart} onMouseMove={(e) => dragX !== 0 || e.buttons === 1 ? handleMove(e) : null} onMouseUp={handleEnd}
        onTouchStart={handleStart} onTouchMove={handleMove} onTouchEnd={handleEnd}
      >
        <div className="flex justify-between items-start mb-2">
          <span className="text-xs font-semibold bg-blue-100 text-blue-700 px-2 py-0.5 rounded-full">
            {current.category}
          </span>
          <span className="text-xs text-gray-400">{current.posted_ago}</span>
        </div>
        <h3 className="font-bold text-base mb-1">{current.title}</h3>
        <p className="text-sm text-gray-600 line-clamp-3">{current.description}</p>
        <div className="flex gap-3 mt-3 text-xs text-gray-500">
          <span>📍 {current.distance_km}km</span>
          <span>💰 {current.budget}</span>
        </div>
        {dragX > 50 && <div className="absolute top-4 right-4 text-green-600 font-bold text-lg rotate-12">ACCEPT</div>}
        {dragX < -50 && <div className="absolute top-4 left-4 text-red-600 font-bold text-lg -rotate-12">PASS</div>}
      </div>

      <div className="absolute bottom-0 inset-x-4 flex justify-center gap-4">
        <button onClick={() => { onDecline(current.id); advance(); }} className="w-12 h-12 rounded-full bg-red-100 text-red-600 text-xl">✕</button>
        <button onClick={() => { onAccept(current.id); advance(); }} className="w-12 h-12 rounded-full bg-green-100 text-green-600 text-xl">✓</button>
      </div>
    </div>
  );
}

function EmptyState() {
  return <div className="text-center text-gray-400 py-12">No new cases right now — check back soon.</div>;
}
