STEP 3 — Create digest sender service
Create artifacts/litigaforge-ai/routers/digest.py

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel, EmailStr
from database import get_db_connection
import uuid, smtplib, os
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from datetime import datetime, date
import logging

router = APIRouter()
logger = logging.getLogger(__name__)

BASE_URL = "https://litigaforge.com"

class SubscribeRequest(BaseModel):
    name: str
    email: EmailStr
    country: str = "in"

@router.post("/digest/subscribe")
async def subscribe(req: SubscribeRequest):
    confirm_token = str(uuid.uuid4())
    unsubscribe_token = str(uuid.uuid4())
    
    async with get_db_connection() as conn:
        existing = await conn.fetchval(
            "SELECT id FROM digest_subscribers 
             WHERE email=$1", req.email
        )
        if existing:
            raise HTTPException(
                status_code=400,
                detail="Email already subscribed"
            )
        await conn.execute("""
            INSERT INTO digest_subscribers 
            (name, email, country, confirm_token, 
             unsubscribe_token)
            VALUES ($1,$2,$3,$4,$5)
        """, req.name, req.email, req.country,
            confirm_token, unsubscribe_token)
    
    # Send confirmation email
    confirm_url = (
        f"{BASE_URL}/api/digest/confirm/"
        f"{confirm_token}"
    )
    await send_confirmation_email(
        req.email, req.name, confirm_url
    )
    
    return {
        "message": "Check your email to confirm 
                    subscription"
    }

@router.get("/digest/confirm/{token}")
async def confirm_subscription(token: str):
    async with get_db_connection() as conn:
        result = await conn.fetchrow(
            """UPDATE digest_subscribers 
               SET confirmed=true 
               WHERE confirm_token=$1 
               RETURNING email, name""",
            token
        )
        if not result:
            raise HTTPException(
                status_code=404,
                detail="Invalid confirmation token"
            )
    return {
        "message": f"Subscription confirmed for 
                    {result['email']}"
    }

@router.get("/digest/unsubscribe/{token}")
async def unsubscribe(token: str):
    async with get_db_connection() as conn:
        result = await conn.execute(
            """UPDATE digest_subscribers 
               SET is_active=false 
               WHERE unsubscribe_token=$1""",
            token
        )
    return {"message": "Successfully unsubscribed"}

@router.post("/admin/digest/send")
async def send_digest_now():
    """Manual trigger to send digest immediately."""
    result = await _send_daily_digest()
    return result

async def _send_daily_digest():
    """Core digest sending function."""
    async with get_db_connection() as conn:
        # Get 5 most recent judgments
        judgments = await conn.fetch("""
            SELECT case_name, court, court_slug,
                   year, area_of_law, outcome,
                   summary_en, slug
            FROM judgments
            WHERE status='published'
            ORDER BY created_at DESC
            LIMIT 5
        """)
        
        # Get active confirmed subscribers
        subscribers = await conn.fetch("""
            SELECT name, email, unsubscribe_token
            FROM digest_subscribers
            WHERE is_active=true AND confirmed=true
        """)
    
    if not subscribers:
        return {"sent": 0, "message": "No subscribers"}
    
    if not judgments:
        return {"sent": 0, "message": "No judgments"}
    
    # Build cases HTML
    today = date.today().strftime("%B %d, %Y")
    cases_html = ""
    
    for j in judgments:
        outcome = j['outcome'] or "Unknown"
        outcome_lower = outcome.lower()
        if "landmark" in outcome_lower:
            outcome_class = "landmark"
        elif "upheld" in outcome_lower or \
             "allowed" in outcome_lower:
            outcome_class = "upheld"
        elif "dismissed" in outcome_lower:
            outcome_class = "dismissed"
        else:
            outcome_class = "other"
        
        case_url = (
            f"{BASE_URL}/in/judgments/"
            f"{j['court_slug']}/{j['year']}/{j['slug']}"
        )
        summary = (j['summary_en'] or "")[:300]
        if len(j['summary_en'] or "") > 300:
            summary += "..."
        
        cases_html += f"""
        <div class="case-card">
          <div class="case-title">{j['case_name']}</div>
          <div class="case-meta">
            {j['court']} · {j['year']} · 
            {j['area_of_law'] or 'General'}
          </div>
          <span class="outcome-badge 
                outcome-{outcome_class}">
            {outcome}
          </span>
          <div class="case-summary">{summary}</div>
          <a href="{case_url}" class="read-more">
            Read Full Analysis →
          </a>
        </div>
        <hr class="divider">
        """
    
    # Load template
    template_path = os.path.join(
        os.path.dirname(__file__),
        "../templates/digest_email.html"
    )
    with open(template_path) as f:
        template = f.read()
    
    html_body = template\
        .replace("{{ date }}", today)\
        .replace("{{ cases_html }}", cases_html)
    
    # Send to all subscribers
    sent = 0
    errors = 0
    
    for sub in subscribers:
        try:
            unsub_url = (
                f"{BASE_URL}/api/digest/unsubscribe/"
                f"{sub['unsubscribe_token']}"
            )
            personalized = html_body.replace(
                "{{ unsubscribe_url }}", unsub_url
            )
            await _send_email(
                to_email=sub['email'],
                to_name=sub['name'],
                subject=f"⚖️ Today's Top Judgments — "
                        f"{today}",
                html_body=personalized
            )
            sent += 1
        except Exception as e:
            logger.error(
                f"Failed to send to {sub['email']}: {e}"
            )
            errors += 1
    
    # Update last_sent_at
    async with get_db_connection() as conn:
        await conn.execute(
            """UPDATE digest_subscribers 
               SET last_sent_at=NOW() 
               WHERE is_active=true AND confirmed=true"""
        )
    
    return {
        "sent": sent, 
        "errors": errors,
        "judgments_included": len(judgments),
        "date": today
    }

async def _send_email(
    to_email: str, 
    to_name: str,
    subject: str, 
    html_body: str
):
    """Send email using existing SMTP config."""
    smtp_host = os.getenv("SMTP_HOST")
    smtp_port = int(os.getenv("SMTP_PORT", "587"))
    smtp_user = os.getenv("SMTP_USER")
    smtp_pass = os.getenv("SMTP_PASSWORD")
    from_email = os.getenv(
        "SMTP_FROM", 
        "digest@litigaforge.com"
    )
    
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = f"LitigaForge Digest <{from_email}>"
    msg["To"] = f"{to_name} <{to_email}>"
    msg.attach(MIMEText(html_body, "html"))
    
    with smtplib.SMTP(smtp_host, smtp_port) as server:
        server.starttls()
        server.login(smtp_user, smtp_pass)
        server.send_message(msg)

async def send_confirmation_email(
    email: str, name: str, confirm_url: str
):
    html = f"""
    <div style="font-family:Arial;max-width:500px;
                margin:0 auto;padding:24px;">
      <h2>⚖️ Confirm your LitigaForge subscription</h2>
      <p>Hi {name},</p>
      <p>Click below to confirm your subscription to 
      the Daily Judgment Digest — 5 new court cases 
      explained every morning.</p>
      <a href="{confirm_url}" 
         style="background:#4f46e5;color:white;
                padding:12px 24px;text-decoration:none;
                border-radius:6px;display:inline-block;
                margin:16px 0;">
        Confirm Subscription
      </a>
      <p style="color:#888;font-size:12px;">
        If you didn't subscribe, ignore this email.
      </p>
    </div>
    """
    await _send_email(
        email, name,
        "Confirm your LitigaForge Daily Digest",
        html
    )
